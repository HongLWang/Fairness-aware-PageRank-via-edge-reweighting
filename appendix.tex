\end{equation}

\begin{equation*}

\end{equation*}
where $\Ematrix_{ij}$ is the matrix whose $[i,j]$ entry is 1 and all other entries are 0.

\section{Approximating $\vecty_k$}
\label{appendix:Neumann series}

In this section, we prove that for approximating $\vecty_k$, truncating the Neumann series to its first $50$ terms yields a relative error below $0.0003$.

According to the Neumann series formula, we can calculate the matrix inverse of a matrix $\textbf{X}$ as: 
$
(\textbf{I} - \textbf{X})^{-1} = \sum_{t=0}^\infty \textbf{X}^t.
$ 
Let $\textbf{S}_k$ denote the partial sum of the Neumann series up to the $k$-th term, that is,
\begin{equation*}
    \textbf{S}_k = \sum_{t=0}^k \textbf{X}^t = \textbf{I} + \textbf{X} + \textbf{X}^2 + \dots + \textbf{X}^k.
\end{equation*}
We denote the truncation error as
$
\textbf{E}_k = (\textbf{I} - \textbf{X})^{-1} - \textbf{S}_k
$, then we can express $\textbf{E}_k$ as the sum of the remaining terms:
\begin{equation*}
    \textbf{E}_k = \sum_{t=k+1}^\infty \textbf{X}^t
\end{equation*}
By factoring out $\textbf{X}^{k+1}$ from the series we get:
\begin{equation*}
    \textbf{E}_k = \textbf{X}^{k+1} \sum_{t=0}^\infty \textbf{X}^t.
\end{equation*}
Using the formula for the Neumann series, the infinite sum $\sum_{t=0}^\infty \textbf{X}^t$ is equal to $(\textbf{I} - \textbf{X})^{-1}$. Thus:
\begin{equation*}
    \textbf{E}_k = \textbf{X}^{k+1} (\textbf{I} - \textbf{X})^{-1}
\end{equation*}
By taking the matrix norm of both sides, we get
\begin{equation*}
    \|\textbf{E}_k\| \leq \|\textbf{X}^{k+1}\| \cdot \|(\textbf{I} - \textbf{X})^{-1}\|
\end{equation*}
Given that $|\textbf{X}|<1$, it follows that
\begin{equation*}
    \|\textbf{E}_k\| \leq \|\textbf{X}\|^{k+1} \cdot \|(\textbf{I} - \textbf{X})^{-1}\|
\end{equation*}
Thus, we conclude that the relative error for approximating $(1-\textbf{X})^{-1}$ is given by $\|\textbf{X}\|^{k+1}$.

% Since $\matrixU^T = 1 - (1 - \gamma) \transmatrix$, to evaluate the approximation loss of $\vecty_k = \gamma \matrixU^{-T} \indvect{k}$, we let $\textbf{X}=(1-\gamma) \transmatrix $, with $\gamma$ set to $0.15$, and consequently, $|(1-\gamma) \transmatrix|_\infty = 0.85$. We conclude that the relative error of using the first $t$ term approximation is $0.85^{t+1}$. By setting $t = 50$ the relative error is smaller than 0.0003.

Since $\matrixU^T = 1 - (1 - \gamma) \transmatrix$, to evaluate the approximation error of $\vecty_k = \gamma \matrixU^{-T} \indvect{k}$, we set $\mathbf{X} = (1-\gamma) \transmatrix$ with $\gamma = 0.15$. Consequently, $\lVert (1-\gamma) \transmatrix \rVert_\infty = 0.85$. It follows that the relative error of the $t$-term approximation is $0.85^{\,t+1}$. Setting $t = 50$ yields a relative error below $0.0003$.

\section{Convergence Analysis for minimizing $\mathcal{L}(\transmatrix)$}
\label{section:convergence_L}
In this section, we first obtain an upper bound for the Lipschitz constant for the loss function, and then prove that the projected gradient descent method converges to a stationary point.

Given restart probability $\gamma$ and restart vector $\vectv$, we can rewrite the loss function as :
\begin{equation*}
    \mathcal{L}(\transmatrix) = \frac{1}{K} \sum_{k=1}^K \left( g_k(\transmatrix) \right)^2
\end{equation*}
where $g_k(\transmatrix) = \indvect{\agroup}^T \prvector - \phi_k$ and $\prvector = \gamma (I - (1-\gamma) \transmatrix^T)^{-1} \vectv.$


\subsection{Gradient of the Loss $\mathcal{L}(\transmatrix)$}

Though we have already provided the first-gradient analysis of the loss function $\mathcal{L}(\transmatrix)$, we restate it here for ease of analysis. We first compute the gradient of \(\mathcal{L}\) with respect to \(\transmatrix_{ij}\):
\begin{equation*}
    \frac{\partial \mathcal{L}}{\partial \transmatrix_{ij}}
= \frac{2}{K} \sum_{k=1}^K g_k \frac{\partial h_k}{\partial \transmatrix_{ij}}
\end{equation*}
where \( h_k(\transmatrix) = \indvect{\agroup}^T \prvector \).

% \paragraph{\textbf{Derivative of \( h_k(P) \) with Respect to \( P_{ij} \)}}

Let \(\Amatrix = I - (1-\gamma) \transmatrix^T\), then it holds that, $\Amatrix \prvector = \gamma \vectv$, and 
\begin{equation*}
    \frac{\partial \Amatrix \prvector}{\partial \transmatrix_{ij}} = \frac{\partial \gamma \vectv}{\partial \transmatrix_{ij}} = 0.
\end{equation*}
Using the product rule, it holds that
\begin{equation*}
    \frac{\partial \Amatrix}{\partial \transmatrix_{ij}} \prvector + \Amatrix \frac{\partial \prvector}{\partial \transmatrix_{ij}} = 0,
\end{equation*}
and thus
\begin{equation*}
    \frac{\partial \prvector}{\partial \transmatrix_{ij}} = -\invAmatrix\frac{\partial \Amatrix}{\partial \transmatrix_{ij}} \prvector.
\end{equation*}
where
\begin{align*}
    \frac{\partial \Amatrix}{\partial \transmatrix_{ij}} &= \frac{\partial I - (1 - \gamma)\partial \transmatrix^T}{\partial \transmatrix_{ij}} \\
    & = -(1-\gamma) \frac{\partial \transmatrix^T}{\partial \transmatrix_{ij}} = -(1 - \gamma) \Ematrix_{ji},
\end{align*}
with $\Ematrix_{ji}$ denoting the matrix where only the $(j,i)$-th element is 1, and all other elements are zero. 

Therefore,
\begin{align*}
\frac{\partial h_k}{\partial \transmatrix_{ij}} &= \indvect{\agroup}^T \frac{\partial \prvector}{\partial \transmatrix_{ij}} \\
&= (1-\gamma) \indvect{\agroup}^T \invAmatrix \Ematrix_{ji} \prvector \\
&= (1-\gamma) \prvector_i (\indvect{\agroup}^T \invAmatrix \matr{e}_j)
\end{align*}
where $\matr{e}_j$ is the standard basis vector (1 at position $j$, zero elsewhere).



\subsection{Second Derivative of the Loss $\mathcal{L}(P)$}

We now compute the second derivative of the loss:
\begin{align*}
\frac{\partial^2 \mathcal{L}}{\partial \transmatrix_{ij} \partial \transmatrix_{ab}}
&= \frac{\partial}{\partial \transmatrix_{ab}} \left( \frac{\partial \mathcal{L}}{\partial \transmatrix_{ij}} \right) \\
&= \frac{\partial}{\partial \transmatrix_{ab}} \left[ \frac{2}{K} \sum_{k=1}^K g_k \frac{\partial h_k}{\partial \transmatrix_{ij}} \right] \\
&= \frac{2}{K} \sum_{k=1}^K \left[
    \frac{\partial g_k}{\partial \transmatrix_{ab}} \frac{\partial h_k}{\partial \transmatrix_{ij}}
    + g_k \frac{\partial^2 h_k}{\partial \transmatrix_{ij} \partial \transmatrix_{ab}}
\right]
\end{align*}
We calculate each term of the second derivative respectively.
\paragraph{First term:}
\[
\frac{\partial g_k}{\partial \transmatrix_{ab}} = \frac{\partial h_k}{\partial \transmatrix_{ab}} = (1-\gamma) \prvector_a (\indvect{\agroup}^T \invAmatrix \matr{e}_b)
\]

\paragraph{Second term:}
\begin{align*}
\frac{\partial^2 h_k}{\partial \transmatrix_{ij} \partial \transmatrix_{ab}}
&= (1-\gamma) \frac{\partial}{\partial \transmatrix_{ab}} \left[ \prvector_i (\indvect{\agroup}^T \invAmatrix \matr{e}_j) \right] \\
&= (1-\gamma) \left[
    \frac{\partial \prvector_i}{\partial \transmatrix_{ab}} (\indvect{\agroup}^T \invAmatrix \matr{e}_j)
    + \prvector_i \frac{\partial}{\partial \transmatrix_{ab}} (\indvect{\agroup}^T \invAmatrix \matr{e}_j)
\right]
\end{align*}
We have:
\[
\frac{\partial \prvector_i}{\partial \transmatrix_{ab}} = (1-\gamma) [\invAmatrix \Ematrix_{ba} \prvector]_i
\]
and
\[
\frac{\partial}{\partial \transmatrix_{ab}} (\indvect{\agroup}^T \invAmatrix \matr{e}_j) = (1-\gamma) \indvect{\agroup}^T \invAmatrix \Ematrix_{ba} \invAmatrix \matr{e}_j
\]
{Combining the two terms, we get}
\begin{align*}
\frac{\partial^2 h_k}{\partial \transmatrix_{ij} \partial \transmatrix_{ab}} &=
(1-\gamma)^2 \Big(
    [\invAmatrix \Ematrix_{ba} \prvector]_i \cdot (\indvect{\agroup}^T \invAmatrix \matr{e}_j)
    + \prvector_i \cdot \indvect{\agroup}^T \invAmatrix \Ematrix_{ba} \invAmatrix \matr{e}_j
\Big)
\end{align*}

The Hessian entry of \(\mathcal{L}(\transmatrix)\) at \(((ij), (ab))\) is:
\begin{align}
\label{eq:hessian}
    & \frac{\partial^2 \mathcal{L}}{\partial \transmatrix_{ij} \partial \transmatrix_{ab}}
= \frac{2}{K} \sum_{k=1}^K \Bigg[
        (1-\gamma)^2 \prvector_a \prvector_i (\indvect{\agroup}^T \invAmatrix \matr{e}_b)(\indvect{\agroup}^T \invAmatrix \matr{e}_j)\\ \nonumber
    &+\ g_k(\transmatrix) \cdot (1-\gamma)^2 \Big(
        [\invAmatrix \Ematrix_{ba} \prvector]_i \cdot (\indvect{\agroup}^T \invAmatrix \matr{e}_j)
        + \prvector_i \cdot \indvect{\agroup}^T \invAmatrix \Ematrix_{ba} \invAmatrix \matr{e}_j
    \Big)
\Bigg]
\end{align}
\subsection{Lipschitz Constant for $\mathcal{L}(P)$}
Since $\mathcal{L}$ is twice differentiable, $\mathcal{L}$ is Lipschitz smooth with constant $L$ if and only if the operator norm of the Hessian matrix $ \frac{\partial^2 \mathcal{L}}{\partial P_{ij} \partial P_{ab}}$ is bounded by $L$ for all $P$ in the domain.
In this section, we flat the Hessian matrix and show an upper bound for its operator norm, which serves as an upper bound to its Lipschitz constant.

In order to flatten the Hessian matrix, let $r \in [n^2]$ be the row indexes and $s \in [n^2]$ be the column indexes. Given any $(i,j)$ pair, we can calculate a unique flattened row index $r = (i-1)n + j$ and given any $(a,b)$ pair, we can calculate a unique flattened column index $s = (a-1)n + b$. Conversely, given any pair $r,s$ we can calculate the unique corresponding for indices $ i(r), j(r), a(s) \textit{ and } b(s)$ as follows:
\begin{align*}
    &i(r) = \lfloor \frac{r-1}{n}\rfloor + 1, j(r) = r - (i(r) -1)n \\
    &a(s) = \lfloor \frac{s-1}{n}\rfloor + 1, b(s) = s - (a(s) -1)n 
\end{align*}

We can then rewrite \Cref{eq:hessian} as
\begin{align*}
\label{eq:hessian}
     \Hmatrix_{rs}
&= \frac{2}{K} \sum_{k=1}^K \Bigg[
        (1-\gamma)^2 \prvector_{a(s)} \prvector_{i(r)}(\indvect{\agroup}^T \invAmatrix \matr{e}_{b(s)})(\indvect{\agroup}^T \invAmatrix \matr{e}_{j(r)})\\ \nonumber
    &+\ g_k(\transmatrix) \cdot (1-\gamma)^2 \Big(
        [\invAmatrix \Ematrix_{b(s)a(s)} \prvector]_{i(r)} \cdot (\indvect{\agroup}^T \invAmatrix \matr{e}_{j(r)} \\ \nonumber 
        &+ \ g_k(\transmatrix) \cdot (1-\gamma)^2 \prvector_{i(r)} \cdot \indvect{\agroup}^T \invAmatrix \Ematrix_{b(s)a(s)} \invAmatrix \matr{e}_{j(r)}
    \Big)
\Bigg]
\end{align*}
Let 
\begin{align*}
    \Hmatrix_{rs}^{(1)} & = \frac{2}{K} (1-\gamma)^2\sum_{k=1}^K 
         \prvector_{a(s)} \prvector_{i(r)}(\indvect{\agroup}^T \invAmatrix \matr{e}_{b(s)})(\indvect{\agroup}^T \invAmatrix \matr{e}_{j(r)}) \\
        \Hmatrix_{rs}^{(2)} & = \frac{2}{K} (1-\gamma)^2\sum_{k=1}^K 
        \ g_k(\transmatrix) 
        [\invAmatrix \Ematrix_{b(s)a(s)} \prvector]_{i(r)} \cdot (\indvect{\agroup}^T \invAmatrix \matr{e}_{j(r)}) \\    
        \Hmatrix_{rs}^{(3)} & = \frac{2}{K} (1-\gamma)^2\sum_{k=1}^K 
        \ g_k(\transmatrix)  \prvector_{i(r)} \cdot \indvect{\agroup}^T \invAmatrix \Ematrix_{b(s)a(s)} \invAmatrix \matr{e}_{j(r)} 
\end{align*}
Then it holds that
\begin{equation*}
     \Hmatrix_{rs} =  \Hmatrix_{rs}^1 +  \Hmatrix_{rs}^2 +  \Hmatrix_{rs}^3
\end{equation*}
To be able to bound the operator norm of $H$, we bound the operator norm of $\Hmatrix^{(1)},  \Hmatrix^{(2)}, \text{and }  \Hmatrix^{(3)}$ separately.


Before bounding the operator norm of each term, we first proof $\|\invAmatrix\|_2 \leq 1/\gamma$, which will be used multiple times in the following deriviation.
 The proof is the result of three observations: (1) the largest eigenvalue of $P^T$ is 1, (2) the smallest eigenvalue of $\Amatrix$ is $\gamma$ and (3) the operator norm of the symmetric matrix $\|\invAmatrix\|_2$ is the reciprocal of the smallest absolute eigenvalue of $\Amatrix$. We provide details for each observation in the following:
\begin{enumerate}
	\item Let $\textbf{1}$ be an all-one vector, i.e. $\textbf{1} = (1,1,...,1)^T$. We can see $\textbf{1}$ is a eigenvector of $\transmatrix$ since $\transmatrix \textbf{1} = 1$. We also know the largest eigenvalue is bounded by the infinity norm, which is $\| \transmatrix\|_{\infty} = \max_i \sum_{j}|\transmatrix_{ij}| = 1$. Thus, the maximum eigenvalue of $P^T$, which is equal to the maximum eigenvalue of $P$, is equal to 1.
	
	\item Let $\lambda$ denote each eigenvalue of $\transmatrix^T$, and $\mu$ each eigenvalue of $\Amatrix$. Since $\Amatrix = I - (1 - \gamma) \transmatrix^T$, it holds that $\mu = 1 - (1 - \gamma) \lambda$. Thus the smallest eigenvalue of $\Amatrix$ occurs when $\lambda = 1$, leading to $u_{min} = 1 - (1 - \gamma) = \gamma$
	
	\item Combining (1) and (2), we get $\| \invAmatrix\| = \frac{1}{\mu_{min}} = \frac{1}{\gamma}$. 
\end{enumerate}

\subsection{Convergence analysis of the projected gradient descent method for minimizing $\mathcal{L}(\transmatrix)$}
Since $\mathcal{L}(P)$ is a differentiable function and  $\Delta \mathcal{L}$ is $L$-Lipschitz, and its feasible set $\resfeasiblematrices(\transmatrix) \cap \feasiblematrices(\transmatrix)$ is convex and closed, the projected gradient update with constant step size $0 \leq \alpha \leq 2/L$ converges to a stationary point, which is a local minimum of $\mathcal{L}$ over $\resfeasiblematrices(\transmatrix) \cap \feasiblematrices(\transmatrix)$ \cite{bertsekas1997nonlinear}.


\section{Convergence Analysis for minimizing $\mathcal{L}_g(\transmatrix)$}
\label{section:convergence_Lg}

We can derive the Hessian matrix of $\mathcal{L}_g(\transmatrix)$ and its Lipschitz constant in a manner analogous to that for $\mathcal{L}(\transmatrix)$. Specifically, let $\textbf{H}_g$ be the Hessian matrix of $\mathcal{L}_g$, then it holds that  
\begin{align}
\label{eq:hessian_adapt}
& \textbf{H}_g[(i,j),(a,b)] = 
     \frac{\partial^2 \mathcal{L}_g}{\partial \transmatrix_{ij} \partial \transmatrix_{ab}}  \\ \nonumber
& = \frac{2}{K^2} \sum_{k=1}^K \sum_{l=1}^K \Bigg[
        (1-\gamma)^2 {\prvector_l}_a {\prvector_l}_i (\indvect{\agroup}^T \invAmatrix \matr{e}_b)(\indvect{\agroup}^T \invAmatrix \matr{e}_j)\\ \nonumber
    &+\ g_k(\transmatrix) \cdot (1-\gamma)^2 \Big(
        [\invAmatrix \Ematrix_{ba} {\prvector_l}]_i \cdot (\indvect{\agroup}^T \invAmatrix \matr{e}_j)
        + {\prvector_l}_i \cdot \indvect{\agroup}^T \invAmatrix \Ematrix_{ba} \invAmatrix \matr{e}_j
    \Big)
\Bigg]
\end{align}
Equation~\eqref{eq:hessian_adapt} follows directly from \Cref{eq:hessian} by replacing $\prvector$ with $\prvector_l$ throughout and introducing an additional summation over $l \in \{1, \ldots, \nogroups\}$.  

The derivation of the Lipschitz constant for $\mathcal{H}$ remains valid for the inner sum of $\mathcal{H}_g$ when $\prvector$ is replaced by $\prvector_l$. Given the additivity property of the operator norm, we conclude that the Lipschitz constant of $\mathcal{L}_g$ is equal to that of $\mathcal{L}_g$, and the projected gradient descent method converges to a local minima with a constant step size $0 \leq \alpha \leq 2/L$.

    \mathbf{R} = \delta_1 \mathbf{x}^T + \delta_2 \mathbf{y}^T
\end{equation}
where $\delta_1[i] = \grouppr_1 -\frac{(1-\grouppr_1)out_1(i)}{out_2(i)}$, $\delta_2[i] = (1-\grouppr_1) -\frac{\grouppr_1 out_2(i)}{out_1(i)}$. $\mathbf{x}$ and $\mathbf{y}$ are two indicator vectors where $\mathbf{x}(i) = 1 \text{ if }  i\in \group_1$ and $\mathbf{y}(i) = 1 \text{ if } i\in \group_2$.


\smallskip
\noindent 
\lfprn \citep{tsioutsiouliklis_fairness-aware_2021}. 
\Amatrix neighborhood-based algorithm that achieves perfect $\groupDist$-fairness in \pagerank scores. The revised transition matrix is given by $\newtransmatrix = \sum_{\agroup =1}^{2}\grouppr_\agroup \transmatrix_\agroup$, where 
\begin{equation}
    \transmatrix_\agroup =  \left\{\begin{matrix}
        \frac{1}{out_\agroup(i)}, &  \text{if } (i,j) \in E \text{ and } j \in \group_k \\
         \frac{1}{|\group_k|}, & \text{if } out_\agroup(i) = 0 \text{ and } j \in \group_k \\
         0 &  \text{otherwise} \\
\end{matrix}\right.
\end{equation}

\smallskip
\noindent 
\fairwalk \citep{rahman_fairwalk_2019}. 
This algorithm reweights the edges such that a random walker has an equal probability of visiting each reachable group and an equal probability of visiting each vertex within each reachable group. \fairwalk does not change the restart vector, and the revised transition matrix is given by:
\begin{equation}
\label{eq:fairwalk}
    \newtransmatrix[i,j] = \frac{\grouppr_{\labelof_j} \transmatrix[i,j]}{\sum_{m \in \group_{\labelof_j}}P[i,m] \sum_{\agroup =1}^{\nogroups} I_{\agroup}(i)\grouppr_\agroup} ,
\end{equation}
where $I_\agroup(i) = 1$ if vertex $i$ has a out-neighbor that belongs to group $\agroup$ and 0 otherwise.


\smallskip
\noindent 
\crosswalk \citep{khajehnejad_crosswalk_2022}. 
This algorithm extends \fairwalk by taking into account the distance of each vertex to other groups in the random walk. \crosswalk upweights edges that (1) are closer to the group boundaries or (2) connect different groups in the graph. This algorithm does not change the restart vector, and we refer readers to Algorithm 1 in the paper for the revised transition matrix.
